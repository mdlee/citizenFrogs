# citizen frogs
# signal detection, many frogs
# dprime and bias as person property
model{
  for (t in 1:nTrials){
    yHat[t, 1] = 1 - phi(d[person[t]]/2 + c[person[t]] )
    yHat[t, 2] = 1 - phi(-d[person[t]]/2 + c[person[t]])
    y[t] ~ dbern(yHat[t, tau[stimulus[t], frog[t]]+1])
  }
  for (j in 1:nFrogs){
    gammaArrive[j] ~ dunif(0, 1)
    gammaLeave[j] ~ dunif(0, 1)
    phi[j] = gammaArrive[j]/(gammaArrive[j]+gammaLeave[j])
    tau[1, j] ~ dbern(0.5)
    for (i in 2:nStimuli){
      tau[i, j] ~ dbern(equals(tau[i-1, j], 1)*(1-gammaLeave[j]) + equals(tau[i-1, j], 0)*gammaArrive[j])
    }
  }
  for (i in 1:nPeople){
    d[i] ~ dnorm(0, 1/3^2)T(0,)
    c[i] ~ dnorm(0, 1/3^2)
  }
}
