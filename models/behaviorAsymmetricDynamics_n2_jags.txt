# citizen frogs
# behavior and asymmetric dynamics
model{
  for (t in 1:nTrials){
    thetaHat[t, 1] = 1 - theta[frog[t]]
    thetaHat[t, 2] = theta[frog[t]]
    y[t] ~ dbern(thetaHat[t, tau[stimulus[t], frog[t]]+1])
  }
  for (j in 1:nFrogs){
    theta[j] ~ dunif(0.5, 1)
    gammaArrive[j] ~ dunif(0, 1)
    gammaLeave[j] ~ dunif(0, 1)
    phi[j] = gammaArrive[j]/(gammaArrive[j]+gammaLeave[j])
    tau[1, j] ~ dbern(0.5)
    for (i in 2:nStimuli){
      tau[i, j] ~ dbern(equals(tau[i-1, j], 1)*(1-gammaLeave[j]) + equals(tau[i-1, j], 0)*gammaArrive[j])
    }
  }
}
