model {
  # --- Observation Model ---
  for (t in 1:nTrials) {
    yHat[t, 1] = beta[person[t]]
    yHat[t, 2] = alpha[person[t]] + (1 - alpha[person[t]]) * beta[person[t]]
    
    # tau is the discrete state of the frog (0 or 1)
    y[t] ~ dbern(yHat[t, tau[stimulus[t], frog[t]] + 1])
  }

  # --- Social Dynamics (Correlated Latent States) ---
  for (i in 2:nStimuli) {
    # Latent vector z determines presence across all frogs at time i
    # dmnorm handles the covariance (social/environmental coupling)
    z[i, 1:nFrogs] ~ dmnorm(muSocial[i, 1:nFrogs], precSocial[1:nFrogs, 1:nFrogs])
    
    for (j in 1:nFrogs) {
      # The "mean" latent state depends on the frog's previous state
      # stepArrive and stepLeave are the probit-scale thresholds
      muSocial[i, j] = equals(tau[i-1, j], 1) * stepLeave[j] + 
                       equals(tau[i-1, j], 0) * stepArrive[j]
      
      # Discrete state: 1 if latent z > 0, else 0
      tau[i, j] = step(z[i, j])
    }
  }

  # --- Initial Conditions & Frog Traits ---
  for (j in 1:nFrogs) {
    tau[1, j] ~ dbern(0.5)
    
    # Individual tendencies on the probit scale
    stepArrive[j] ~ dnorm(0, 1)
    stepLeave[j]  ~ dnorm(0, 1)
  }

  # --- Covariance Structure ---
  # Prior for the precision matrix (Inverse of Covariance)
  # RMatrix is a diagonal identity matrix provided in the data
  precSocial[1:nFrogs, 1:nFrogs] ~ dwish(I, nFrogs + 1)
  
  # Derived covariance matrix for inspection
  covSocial[1:nFrogs, 1:nFrogs] = inverse(precSocial[ , ])

  # --- Person Parameters ---
  for (p in 1:nPeople) {
    alpha[p] ~ dunif(0, 1)
    beta[p]  ~ dunif(0, 1)
  }
}
